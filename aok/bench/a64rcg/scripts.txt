def stream_words(words, narrow_words):
    """ How a spec's words go in the stream, which the translator's
    rcg_pack_words() must match: a list of stream words, each a tuple of
    the words it holds. narrow_words are the guest's words that always fit
    in 32 bits, signed. A gadget that accesses memory, which is one with a
    tlb word, has its narrow words two to a stream word, the first in the
    low half, then an odd one out alone, then each of the others alone. Any
    other gadget has each word alone. A memory gadget splits the pairs
    while it waits for the TLB entry; packing the others only made them
    slower. """
    if "tlb" not in words:
        return [(w,) for w in words]
    narrow = [w for w in words if w in narrow_words]
    return ([tuple(narrow[i:i + 2]) for i in range(0, len(narrow), 2)] +
            [(w,) for w in words if w not in narrow_words])

def check_lines(name, lines, harness, operands):
    """ Checks the rendered lines of gadget name, from harness: that each
    names only the asm operands of operands and the registers a body may
    use, none of V8 to V15, which TCTI expects unchanged, and that every
    branch it makes has a label to go to. """
    h = harness
    allowed = {*h.scratch, h.env, h.retaddr, "x15", "sp", "lr", "xzr"}
    labels = {line[:-1] for line in lines if re.fullmatch(r"\\d+:", line)}
    for line in lines:
        if line.startswith("//"):
            continue
        for operand in re.findall(r"%\\[(\\w+)\\]", line):
            assert operand in operands, (name, line)
        # The getpc register may appear, but only as a base address.
        regs = line.replace(f"[{h.getpc}]", "")
        for reg in re.findall(r"\\b([xw](?:\\d+|zr)|sp|lr)\\b", regs):
            assert "x" + reg[1:] in allowed or reg in allowed, (name, line)
        for reg in re.findall(r"(?<![.%])\\b[vqdsh](\\d+)\\b", line):
            assert not 8 <= int(reg) <= 15, (name, line)
        for target in re.findall(r"(?<!\\.)\\b(\\d+)[bf]\\b", line):
            assert target in labels, (name, line)

def write_gadgets(outdir, files, includes, operands, gadgets):
    """ Writes TCTI's gadgets as naked functions to outdir, in a source file
    rcg_gadgets_<file>.c for each name in files, split so that they compile
    in parallel. includes are the headers the files need between osdep.h
    and tcti-gadget-defs.h, operands the "i" asm operands the bodies name,
    and gadgets gives each gadget's file, name and lines. Returns how many
    instructions the gadgets have. """
    operands = ", ".join(f'[{k}] "i" ({v})' for k, v in operands.items())
    out = {}
    for name in files:
        f = open(os.path.join(outdir, f"rcg_gadgets_{name}.c"), "w")
        print("/* Automatically generated by rcg-gen.py. Do not edit. */\\n",
              file=f)
        print('#include "qemu/osdep.h"', file=f)
        for header in includes:
            print(f'#include "{header}"', file=f)
        print('#include "tcti-gadget-defs.h"\\n', file=f)
        out[name] = f

    instructions = 0
    for file, name, lines in gadgets:
        f = out[file]
        instructions += sum(1 for line in lines
                            if not line.endswith(":") and
                            not line.startswith("."))
        print(f"void rcg_gadget_{name}(void);", file=f)
        print(f"__attribute__((naked)) void rcg_gadget_{name}(void)",
              file=f)

        for line in lines:
            print(f'        "{line}\\\\n"', file=f)
        print(f"        :: {operands});\\n}}\\n", file=f)
    for f in out.values():
        f.close()
    return instructions

==> a64/ctl2-patch.py
#!/usr/bin/env python3
# ctl2-patch.py GEN: add control hooks to a copy of the A64 rcg-gen.py (never to the tree's). RCG_CTL in the
# environment then breaks one thing when the gadgets are generated:
#   res:KIND    the result of KIND's gadgets (not MOV/MVN) is wrong in bit 0 when the first register read
#               holds 0x0123456789abcdef, one of the test's values
#   flags       ZF is wrong after any S form whose first register read holds that value
#   movi        the constant gadget stores 0x1235 for 0x1234
#   adr         ADR to the test slot + 4, and ADRP to slot page + 0xfffff000, are wrong in bit 1
#   movzn       MOV and MVN of x28 to x30 are wrong in bit 0
import sys
path = sys.argv[1]
s = open(path).read()
MAGIC = ('"mov x12, #0xcdef", "movk x12, #0x89ab, lsl #16", '
         '"movk x12, #0x4567, lsl #32", "movk x12, #0x0123, lsl #48"')
def rep(old, new):
    global s
    assert s.count(old) == 1, old
    s = s.replace(old, new)
rep('''import argparse
''', '''import argparse
CTL = __import__("os").environ.get("RCG_CTL", "")
CTL_WHAT, _, CTL_KIND = CTL.partition(":")


def ctl_on(spec):
    return (CTL_WHAT == "flags" or (CTL_WHAT == "res" and
            spec.kind == CTL_KIND)) and not getattr(spec, "zn", False)

''')
rep('''        self.emit(f"ldr {dest}, [{self.env}, {self.word(word)}]")
        self.define(dest)

    def write_reg(self, value, word):''', '''        self.emit(f"ldr {dest}, [{self.env}, {self.word(word)}]")
        self.define(dest)
        if ctl_on(self.spec) and not getattr(self, "first", None):
            self.first = dest
            self.emit(f"mov x13, {dest}")

    def write_reg(self, value, word):''')
rep('''        self.emit(f"str {value}, [{self.env}, {self.word(word)}]")
''', '''        if CTL_WHAT == "res" and getattr(self, "first", None):
            skip, _ = self._labels()
            self.emit(''' + MAGIC + ''', "eor x12, x12, x13",
                      f"cbnz x12, {skip}f", f"eor {value}, {value}, #1",
                      f"{skip}:")
        self.emit(f"str {value}, [{self.env}, {self.word(word)}]")
''')
rep('''                  f"cset {w(ADDR)}, ne",
''', '''                  f"cset {w(ADDR)}, ne",
                  *([''' + MAGIC + ''', "cmp x12, x13", "cset w12, eq",
                     f"eor {w(ADDR)}, {w(ADDR)}, w12"]
                    if CTL_WHAT == "flags" and getattr(self, "first", None)
                    else []),
''')
rep('''    g.write_reg(g.word("imm"), "d")
''', '''    if CTL_WHAT == "movi":
        g.emit("mov x12, #0x1234", f"eor x12, x12, {g.word('imm')}",
               "cbnz x12, 1f", f"eor {g.word('imm')}, {g.word('imm')}, #1",
               "1:")
    g.write_reg(g.word("imm"), "d")
''')
rep('''           f"add {A}, {A}, {g.word('imm')}")
    g.write_reg(A, "d")
''', '''           f"add {A}, {A}, {g.word('imm')}")
    if CTL_WHAT == "adr":
        slot = int(CTL_KIND, 16)
        for v in (slot + 4, (slot & ~0xfff) + 0xfffff000):
            g.emit(f"mov x12, #{v & 0xffff:#x}",
                   f"movk x12, #{(v >> 16) & 0xffff:#x}, lsl #16",
                   f"movk x12, #{(v >> 32) & 0xffff:#x}, lsl #32",
                   f"eor x12, x12, {A}", "cmp x12, #0",
                   f"eor x12, {A}, #2", f"csel {A}, x12, {A}, eq")
    g.write_reg(A, "d")
''')
rep('''    g.emit(f"{logic_name(spec.op, spec.inv, spec.flags)} {r}, {a}, {b}")
''', '''    g.emit(f"{logic_name(spec.op, spec.inv, spec.flags)} {r}, {a}, {b}")
    if CTL_WHAT == "movzn" and getattr(spec, "zn", False):
        g.emit("mov x13, %[x28]", f"cmp {g.word('d')}, %[x30]",
               f"ccmp {g.word('m')}, x13, #0, eq",
               f"eor x12, {R}, #1", f"csel {R}, x12, {R}, eq")
''')
rep('''    "pc": "offsetof(CPUARMState, pc)",
''', '''    "pc": "offsetof(CPUARMState, pc)",
    "x28": "offsetof(CPUARMState, xregs[28])",
    "x30": "offsetof(CPUARMState, xregs[30])",
''')
open(path, "w").write(s)

# ctl3-patch.py GEN: add a control hook for the conditional gadgets to a copy of the A64 rcg-gen.py (never to
# the tree's). RCG_CTL=KIND in the environment then makes KIND's gadgets (bcond, cbz, tbz, fused, csel, ccmp)
# take their condition the wrong way when the guest's X28 holds 0x1d1d1d1d1d1d1d39, which is what the test's
# first start puts there and nothing in the test program's own code does.
import sys
path = sys.argv[1]
s = open(path).read()
def rep(old, new):
    global s
    assert s.count(old) == 1, old
    s = s.replace(old, new)
rep('''import argparse
''', '''import argparse
CTL = __import__("os").environ.get("RCG_CTL", "")


def ctl_arm(g, spec, arm):
    """ arm, or with the control on for spec's kind, a condition that is
    arm's the wrong way round when X28 holds the control's value. """
    if spec.kind != CTL:
        return arm
    g.emit(f"cset w13, {arm}", f"ldr x12, [{g.env}, %[x28]]",
           "mov x11, #0x1d39", "movk x11, #0x1d1d, lsl #16",
           "movk x11, #0x1d1d, lsl #32", "movk x11, #0x1d1d, lsl #48",
           "cmp x12, x11", "cset w12, eq", "eor w13, w13, w12",
           "cmp w13, #0")
    return "ne"

''')
rep('''    arm = g.test_cond(spec.cond)
    g.emit(f"{select_name(spec.op)}''', '''    arm = ctl_arm(g, spec, g.test_cond(spec.cond))
    g.emit(f"{select_name(spec.op)}''')
rep('''    arm = g.test_cond(spec.cond)
    g.emit(f"csel {R}, {R}, {g.word('mask')}, {arm}",''', '''    arm = ctl_arm(g, spec, g.test_cond(spec.cond))
    g.emit(f"csel {R}, {R}, {g.word('mask')}, {arm}",''')
rep('''    if spec.jumps:
        g.jumps(arm)
''', '''    arm = ctl_arm(g, spec, arm)
    if spec.jumps:
        g.jumps(arm)
''')
rep('''    "pc": "offsetof(CPUARMState, pc)",
''', '''    "pc": "offsetof(CPUARMState, pc)",
    "x28": "offsetof(CPUARMState, xregs[28])",
''')
open(path, "w").write(s)

   perl -e 'alarm 180; exec @ARGV' $Q -chardev file,path=out.ctl,id=output -kernel rcg-insns; x=$?
   perl -e 'alarm 180; exec @ARGV' $Q -d nochain -chardev file,path=branch.ctl,id=output -kernel rcg-branch; y=$?

   echo "$c, nochain: exit $y, $(wc -l < branch.ctl | tr -d ' ') lines, $(wc -l < differ.$c.branch | tr -d ' ') differ")
done
cp /tmp/rcg-gen.good $G
ninja -C build-a64 > build-a64/ninja-ctl.log 2>&1 && echo "restored and rebuilt" || echo "restore build FAILED"
echo CTL-DONE

#!/bin/bash
# x86's RCG tests on the A64 skeleton's build, as CI runs them: rcg-insns against rcg-insns.ref and rcg-faults,
# with RCG ordered and plain, tracing rcg_insn (now in accel/tcg's trace events), then --coverage.

cd ~/rcgtests-x86 || exit 1
CPU=qemu64,+xsave,+xsaveopt,+avx,+ssse3,+sse4.1,+avx2,+adx,+bmi1,+bmi2,+abm,+fma
run() { # run NAME ACCEL TEST
  local n=$1 a=$2 t=$3 s=$(date +%s)
  $B/qemu-system-x86_64-unsigned -accel tcg,$a -cpu $CPU -monitor none -display none -chardev file,path=$t.$n.out,id=output \
    -device isa-debugcon,chardev=output -device isa-debug-exit,iobase=0xf4,iosize=0x4 \
    -trace rcg_insn -trace rcg_jumps -D trace.$n.$t -kernel $t
  echo "$t $n ($a) exit $? in $(( $(date +%s) - s ))s, $(grep -c rcg_insn trace.$n.$t) rcg_insn events"
}
rm -f *.out trace.*
run ord tcti=on,rcg=on rcg-insns & run ord tcti=on,rcg=on rcg-faults &
run plain tcti=on,rcg=on,tcti-ordered=off rcg-insns & run plain tcti=on,rcg=on,tcti-ordered=off rcg-faults &
wait

  cmp -s rcg-insns.$n.out rcg-insns.ref && echo "$n: rcg-insns matches rcg-insns.ref ($(wc -l < rcg-insns.ref) lines)" || echo "$n: rcg-insns DIFFERS from rcg-insns.ref ($(diff rcg-insns.$n.out rcg-insns.ref | grep -c '^<') lines)"
  echo "  rcg-faults $n: $(tail -1 rcg-faults.$n.out)"

nm $B/qemu-system-x86_64-unsigned > nm.x86
for o in ordered:ord plain:plain; do
  python3 ~/aok-vm-bench/src-a64/target/i386/tcg/rcg-gen.py --coverage ${o%%:*} nm.x86 trace.${o#*:}.* 2>&1 | tail -4
done

==> a64/alp/boot.py
#!/usr/bin/env python3
# boot.py QEMU ACCEL OUT: boot AOK VM's Alpine aarch64 machine (a snapshot of its disk) under ACCEL, run a fixed
# set of commands on its serial console, and write what they print to OUT. For comparing RCG on against off.
import os, re, socket, subprocess, sys, time
qemu, accel, out = sys.argv[1:]
D = os.path.expanduser("~/rcgtests-a64/alpine")
sock = f"/tmp/rcg-a64-{os.getpid()}.sock"
q = subprocess.Popen([qemu, "-accel", f"tcg,{accel}", "-M", "virt", "-cpu", "max", "-smp", "2", "-m", "1024",
    "-kernel", f"{D}/vmlinuz", "-initrd", f"{D}/initramfs", "-append", "root=/dev/vda rw console=ttyAMA0 quiet",
    "-drive", f"file={D}/disk.qcow2,format=qcow2,if=none,id=d0,snapshot=on", "-device", "virtio-blk-pci,drive=d0",
    "-device", "virtio-rng-pci", "-display", "none", "-monitor", "none", "-serial", f"unix:{sock},server,nowait"],
    stdout=subprocess.DEVNULL, stderr=subprocess.STDOUT)
for _ in range(100):
    if os.path.exists(sock): break
    time.sleep(0.1)
ser = socket.socket(socket.AF_UNIX); ser.connect(sock); ser.settimeout(0.5)
buf = b""; raw = open(out + ".raw", "wb")
def read_until(pat, timeout):
    global buf
    end = time.time() + timeout; rx = re.compile(pat.encode())
    while time.time() < end:
        m = rx.search(buf)
        if m:
            got = buf[:m.end()]; buf = buf[m.end():]; return got.decode(errors="replace")
        try:
            d = ser.recv(65536)
            if d: raw.write(d); raw.flush(); buf += d
        except socket.timeout: pass
    q.kill(); raise SystemExit(f"TIMEOUT waiting for {pat!r}; tail: {buf[-300:]!r}")
t0 = time.time()
ser.sendall(b"\n"); read_until(r"(~|/)[^\n]*# ", 600)
boot = time.time() - t0
ser.sendall(b"stty -echo cols 200; export PS1='# '\n"); read_until(r"# $", 30); time.sleep(0.5)
CMDS = [
    "uname -m",
    "sha256sum /bin/busybox /lib/ld-musl-aarch64.so.1",
    "find /etc /bin /sbin /lib -xdev -type f | sort | xargs sha256sum | sha256sum",
    "awk 'BEGIN{s=0;for(i=0;i<300000;i++){s+=(i*i)%7-(i%13)};print s}'",
    "seq 1 20000 | sort -rn | md5sum",
    "i=0; s=0; while [ $i -lt 20000 ]; do s=$((s+i*3-i/7)); i=$((i+1)); done; echo $s",
    "dd if=/dev/zero bs=1M count=8 2>/dev/null | gzip -c | md5sum",
    "expr 9223372036854775807 - 1; echo $((0x7fffffff + 1)) $((-5 - 7)) $((1 - 4294967296))",
]
res = []; n = 0
for c in CMDS:
    n += 1; mark = f"@@{n}@@"
    ser.sendall((c + f"; echo {mark[:2]}''{mark[2:]} $?\n").encode())
    got = read_until(re.escape(mark) + r" (\d+)", 600)
    res.append(f"$ {c}\n{got.strip()}\n")
total = time.time() - t0
open(out, "w").write("\n".join(l.replace("\r", "") for l in res))
ser.sendall(b"poweroff\n")
try: q.wait(60)
except subprocess.TimeoutExpired: q.kill()
os.unlink(sock) if os.path.exists(sock) else None
print(f"{accel}: boot to prompt {boot:.1f}s, total {total:.1f}s, {len(res)} commands")
