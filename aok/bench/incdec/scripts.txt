#!/bin/bash
# inc/dec for RCG (tctiSH/qemu#7): tctiSH's structure-a fed47ac5dd against emkey1/qemu rcg-incdec 3cc89a22c1 (the
# PR, on it), both with rcg=on, and TCTI (structure-a with rcg=off) beside them; M3's flags (../m3/scripts.txt).
export PATH=/opt/homebrew/bin:$PATH GIT_CONFIG_COUNT=1 GIT_CONFIG_KEY_0=url.https://github.com/qemu/.insteadOf GIT_CONFIG_VALUE_0=https://gitlab.com/qemu-project/
cd ~/aok-vm-bench
B="--target-list=x86_64-softmmu --enable-fdt --enable-tcg --disable-werror --disable-docs --enable-slirp --disable-dbus-display --enable-tcg-threaded-interpreter"
git -C src-aok fetch -q emkey1 && git -C src-aok fetch -q https://github.com/tctiSH/qemu structure-a
for v in sa:fed47ac5dd incdec:3cc89a22c1; do
  n=${v%%:*} c=${v#*:}
  [ -d src-$n ] || git -C src-aok worktree add -q --detach ../src-$n $c
  if [ ! -f build-$n/build.ninja ]; then rm -rf build-$n; mkdir build-$n; (cd build-$n && ../src-$n/configure $B > configure.log 2>&1) || { echo "$n: configure FAILED"; tail -15 build-$n/configure.log; exit 1; }; fi
  ninja -C build-$n > build-$n/ninja.log 2>&1 && echo "build-$n ok ($(grep -c warning: build-$n/ninja.log) warnings)" || { echo "build-$n FAILED"; grep -m8 error build-$n/ninja.log; exit 1; }
done
mkdir -p ~/m0-q/sa ~/m0-q/saoff ~/m0-q/incdec
for n in sa incdec; do
cat > ~/m0-q/$n/qemu-system-x86_64 <<W
#!/bin/bash
# rcg=on appended to the -accel bench.py gives.
a=(); while [ \$# -gt 0 ]; do if [ "\$1" = -accel ]; then a+=("\$1" "\$2,rcg=on"); shift 2; else a+=("\$1"); shift; fi; done
exec /Users/mke/aok-vm-bench/build-$n/qemu-system-x86_64-unsigned "\${a[@]}"
W
done
cat > ~/m0-q/saoff/qemu-system-x86_64 <<'W'
#!/bin/sh
exec /Users/mke/aok-vm-bench/build-sa/qemu-system-x86_64-unsigned "$@"
W
chmod +x ~/m0-q/sa/qemu-system-x86_64 ~/m0-q/saoff/qemu-system-x86_64 ~/m0-q/incdec/qemu-system-x86_64
# (bench.py wants a pc-bios beside the binary it is given)
ln -sfn ~/aok-vm-bench/build-sa/pc-bios ~/m0-q/sa/pc-bios; ln -sfn ~/aok-vm-bench/build-sa/pc-bios ~/m0-q/saoff/pc-bios
ln -sfn ~/aok-vm-bench/build-incdec/pc-bios ~/m0-q/incdec/pc-bios
echo INCDEC-BUILT

#!/bin/bash
# The timing: TCTI (structure-a, rcg=off), structure A (rcg=on), and the PR (rcg=on), by wall clock, ROUNDS
# rounds alternating, each after a quiet wait.
R=${1:-3}; cd ~/tctish-bench; mkdir -p ~/incdec-runs
for r in $(seq $R); do
  for c in saoff sa incdec; do
    echo "$(date +%H:%M) round $r $c"
    python3 utils/bench/bench.py run --backend tcti --rounds 1 --label incdec-$c --qemu $HOME/m0-q/$c/qemu-system-x86_64 \
      --wait-quiet 1.5 > $HOME/incdec-runs/time-$c-$r.log 2>&1 || echo "  failed: $(tail -2 $HOME/incdec-runs/time-$c-$r.log)"
  done
done
python3 utils/bench/bench.py compare incdec-saoff incdec-sa incdec-incdec
echo TIME-DONE

#!/bin/bash
# xz again, alone: six rounds, the two rcg=on columns alternating (it is bimodal, about 10 s and 11.6 s, on every build).
cd ~/tctish-bench
for r in 1 2 3 4 5 6; do for c in sa incdec; do
  python3 utils/bench/bench.py run --backend tcti --rounds 1 --workloads xz --label incdec-xz-$c --qemu $HOME/m0-q/$c/qemu-system-x86_64 \
    --wait-quiet 1.5 > $HOME/incdec-runs/xz-$c-$r.log 2>&1 || echo failed $c $r
done; done
python3 utils/bench/bench.py compare incdec-xz-sa incdec-xz-incdec
