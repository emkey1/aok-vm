# === a10x2.sh
#!/bin/bash
# A10X: the new engine against build 12's, alternating, on what TCTI runs: 68k (few gadgets), RISC-V with QEMU_AOK=0; and RISC-V with the gadgets.
U=5c56dee8545a5a555e052ecd50aa0e3ca9334a3b
S=/private/tmp/claude-501/-Users-mke-git-ish-AOK/7b3fb99e-2b30-444a-9291-e34744be360f/scratchpad/a10x
NEW="$HOME/aok-vm-work/dd/Build/Products/Release-iphoneos/AOK VM.app"
OLD="$HOME/aok-vm-work/archives/AOKVM-1.0-12.xcarchive/Products/Applications/AOK VM.app"
SV=/Users/mke/git/aok-vm/aok/app/start-vm.sh
temp() { idevicediagnostics -u $U ioregentry AppleARMPMUCharger 2>/dev/null | grep -A1 -E '<key>Temperature</key>' | tail -1 | sed 's/[^0-9]//g'; }
shot() { # shot NAME
  xcrun devicectl device capture screenshot --device $U --destination "$S/$1.png" > /dev/null 2>&1
}
round() { # round TAG APP N
  local t=$1 app=$2 n=$3
  echo "$(date +%T) install $t (temp $(temp))"
  xcrun devicectl device install app --device $U "$app" 2>&1 | grep -E 'ERROR' | cut -c1-120
  for i in 1 2; do
    bash $SV $U "AOK 68k Console" > /dev/null 2>&1; sleep 80; shot ab-$n-$t-68k$i
    sips --cropToHeightWidth 300 1500 --cropOffset 20 0 "$S/ab-$n-$t-68k$i.png" --out "$S/c-ab-$n-$t-68k$i.png" > /dev/null 2>&1
  done
  AOKVM_ENV='"QEMU_AOK":"0"' bash $SV $U "AOK RISC-V Console" > /dev/null 2>&1; sleep 170; shot ab-$n-$t-rvoff
  bash $SV $U "AOK RISC-V Console" > /dev/null 2>&1; sleep 85; shot ab-$n-$t-rvon
  echo "$(date +%T) done $t (temp $(temp), battery $(ideviceinfo -u $U -q com.apple.mobile.battery -k BatteryCurrentCapacity 2>/dev/null))"
}
round new "$NEW" 1
round old "$OLD" 2
round new "$NEW" 3
bash $SV $U --stop 2>&1 | head -1
rm -rf "$S/crash"; mkdir -p "$S/crash"; idevicecrashreport -k -u $U "$S/crash" > /dev/null 2>&1
find "$S/crash" -type f \( -name 'AOK*' -o -name 'JetsamEvent*' -o -name 'forceReset*' \) -newermt '2026-11-03 00:00' | sed 's|.*/||' | head -10
echo AB-DONE

# A10X: is RISC-V coremark's slow mode under the gadgets in both engines? Four boots of each; the new build stays installed.
U=5c56dee8545a5a555e052ecd50aa0e3ca9334a3b

boots() { # boots TAG
  for i in 1 2 3 4; do

  done

}
boots new
xcrun devicectl device install app --device $U "$OLD" 2>&1 | grep ERROR | cut -c1-120
boots old
xcrun devicectl device install app --device $U "$NEW" 2>&1 | grep ERROR | cut -c1-120
bash $SV $U --stop 2>&1 | head -1
cd "$S" && ./ocr cm-new-*.png cm-old-*.png | cut -c1-210
echo CM-DONE

# === a10x/ocr.swift: RESULT lines from the screenshots, by Vision
import Foundation
import Vision
import AppKit
for path in CommandLine.arguments.dropFirst() {
    guard let img = NSImage(contentsOfFile: path), let cg = img.cgImage(forProposedRect: nil, context: nil, hints: nil) else { print("\(path): no image"); continue }
    let req = VNRecognizeTextRequest()
    req.recognitionLevel = .accurate
    req.usesLanguageCorrection = false
    try? VNImageRequestHandler(cgImage: cg, options: [:]).perform([req])
    let lines = (req.results ?? []).compactMap { $0.topCandidates(1).first?.string }.filter { $0.hasPrefix("RESULT") || $0.contains("STRESS") || $0.contains("checks wrong") || $0.hasPrefix("BENCH-END") }
    print((path as NSString).lastPathComponent, "|", lines.joined(separator: " ; "))
}
